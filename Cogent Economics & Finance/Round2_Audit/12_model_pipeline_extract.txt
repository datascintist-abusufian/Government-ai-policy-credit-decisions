 113: def calibration_intercept_slope(y, p):
 114:     """

 116:         logit(Y) = intercept + slope * logit(p)

 119:         intercept = 0

 123:     x = logit(p).reshape(-1, 1)

 126:         penalty=None,
 127:         solver="lbfgs",
 128:         max_iter=2000
 129:     )

 132:         x,
 133:         y
 134:     )

 137:         float(model.intercept_[0]),
 138:         float(model.coef_[0, 0])

 142: def ece_equal_width(y, p, n_bins=10):
 143:     """

 145:     probability bins.

 158:         p,
 159:         edges[1:-1],
 160:         right=False

 204:             "event_rate": event_rate,

  63: # ------------------------------------------------------------
  64: # B. Decision metrics
  65: # ------------------------------------------------------------
  66: 
  67: def decision_metrics(y, p, threshold, fn_cost, fp_cost=1.0):
  68: 
  69:     flagged = p >= threshold
  70: 
  71:     tp = int(
  72:         np.sum(
  73:             flagged & (y == 1)
  74:         )
  75:     )
  76: 
  77:     fp = int(
  78:         np.sum(
  79:             flagged & (y == 0)
  80:         )
  81:     )
  82: 
  83:     tn = int(
  84:         np.sum(
  85:             (~flagged) & (y == 0)
  86:         )
  87:     )
  88: 
  89:     fn = int(
  90:         np.sum(
  91:             (~flagged) & (y == 1)
  92:         )
  93:     )
  94: 
  95:     n = len(y)
  96: 
  97:     positives = (
  98:         tp + fn
  99:     )
 100: 
 101:     flagged_n = (
 102:         tp + fp
 103:     )
 104: 
 105:     fnr = (
 106:         fn / positives
 107:         if positives > 0
 108:         else np.nan
 109:     )
 110: 
 111:     tpr = (
 112:         tp / positives
 113:         if positives > 0
 114:         else np.nan
 115:     )
 116: 
 117:     review_rate = (
 118:         flagged_n / n
 119:     )
 120: 
 121:     precision = (
 122:         tp / flagged_n
 123:         if flagged_n > 0
 124:         else np.nan
 125:     )
 126: 
 127:     cost = (
 128:         fn_cost * fn
 129:         +
 130:         fp_cost * fp
 131:     )
 132: 
 133:     no_review_cost = (
 134:         fn_cost
 135:         * positives
 136:     )
 137: 
 138:     net_benefit = (
 139:         no_review_cost
 140:         -
 141:         cost
 142:     )
 143: 
 144:     return {
 145:         "threshold": threshold,
 146:         "tp": tp,
 147:         "fp": fp,
 148:         "tn": tn,
 149:         "fn": fn,
 150:         "fnr": fnr,
 151:         "tpr": tpr,
 152:         "review_rate": review_rate,
 153:         "precision": precision,
 154:         "cost": cost,
 155:         "no_review_cost": no_review_cost,
 156:         "net_benefit": net_benefit
 157:     }

 160: # ------------------------------------------------------------
 161: # C. Exact validation threshold search
 162: #
 163: # For a threshold classifier, predictions change only when the
 164: # threshold crosses an observed predicted probability.
 165: #
 166: # We therefore evaluate all unique validation probabilities,
 167: # plus boundary operating points.
 168: # ------------------------------------------------------------
 169: 
 170: def find_cost_minimising_threshold(
 171:     y,
 172:     p,
 173:     fn_cost,
 174:     fp_cost=1.0
 175: ):
 176: 
 177:     order = np.argsort(
 178:         -p,
 179:         kind="mergesort"
 180:     )
 181: 
 182:     p_sorted = p[order]
 183:     y_sorted = y[order]
 184: 
 185:     total_pos = int(
 186:         y_sorted.sum()
 187:     )
 188: 
 189:     total_neg = int(
 190:         len(y_sorted)
 191:         -
 192:         total_pos
 193:     )
 194: 
 195:     cum_tp = np.cumsum(
 196:         y_sorted == 1
 197:     )
 198: 
 199:     cum_fp = np.cumsum(
 200:         y_sorted == 0
 201:     )
 202: 
 203:     # Last position for each unique probability
 204:     is_last_of_value = np.r_[
 205:         p_sorted[:-1] != p_sorted[1:],
 206:         True
 207:     ]
 208: 
 209:     positions = np.flatnonzero(
 210:         is_last_of_value
 211:     )
 212: 
 213:     thresholds = p_sorted[
 214:         positions
 215:     ]
 216: 
 217:     tp = cum_tp[
 218:         positions
 219:     ]
 220: 
 221:     fp = cum_fp[
 222:         positions
 223:     ]
 224: 
 225:     fn = (
 226:         total_pos
 227:         -
 228:         tp
 229:     )
 230: 
 231:     costs = (
 232:         fn_cost * fn
 233:         +
 234:         fp_cost * fp
 235:     )
 236: 
 237:     best_cost = costs.min()
 238: 
 239:     best_positions = np.flatnonzero(
 240:         costs == best_cost
 241:     )
 242: 
 243:     # Tie rule:
 244:     # choose the HIGHEST threshold among equal-cost solutions,
 245:     # i.e. the lower review burden.
 246:     best_local = best_positions[
 247:         np.argmax(
 248:             thresholds[
 249:                 best_positions
 250:             ]
 251:         )
 252:     ]
 253: 
 254:     best_threshold = float(
 255:         thresholds[
 256:             best_local
 257:         ]
 258:     )
 259: 
 260:     return (
 261:         best_threshold,
 262:         float(best_cost)
 263:     )
